
let historialActual = JSON.parse(localStorage.getItem('deposito_gilera_movimientos')) || [];

const destinosPosibles = ["Producción", "Sucursal Puma", "Cierre de Lote"];
const modelosPosibles = ["Smash 110", "Sahel 150", "VC 150", "SMX 200"];

// 2. creamos registros
for(let i = 1; i <= 50; i++) {
  let destinoRandom = destinosPosibles[Math.floor(Math.random() * destinosPosibles.length)];
  let modeloRandom = modelosPosibles[Math.floor(Math.random() * modelosPosibles.length)];
  
  historialActual.push({
    fecha: new Date().toLocaleString('es-AR'), // Usa la fecha y hora de hoy
    modelo: modeloRandom,
    orden: "TEST-OUT-" + (2000 + i),
    factura: "5555" + i,
    cantidad: Math.floor(Math.random() * 40) + 5, // Entre 5 y 45 motos
    destino: destinoRandom
  });
}

// 3. Lo guardamos en la memoria del navegador
localStorage.setItem('deposito_gilera_movimientos', JSON.stringify(historialActual));

// 4. Recargamos
location.reload();

// limpiamos pantalla
localStorage.removeItem('deposito_gilera_movimientos');
